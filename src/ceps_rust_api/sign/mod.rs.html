1//! Local PEM keyring (never accepted from HTTP bodies).
2//!
3//! Values may be inline PEM text or a filesystem path to a `.pem` file.
4
5use crate::error::ApiError;
6use parking_lot::RwLock;
7use std::collections::HashMap;
8use std::path::Path;
9use std::sync::Arc;
10
11#[derive(Clone, Default)]
12pub struct LocalKeyring {
13    inner: Arc<RwLock<HashMap<String, String>>>,
14}
15
16impl LocalKeyring {
17    #[must_use]
18    pub fn new() -> Self {
19        Self::default()
20    }
21
22    /// Parse keyring JSON. `env_name` is used in error messages (`LOCAL_KEYS_JSON` / production).
23    pub fn from_json(raw: &str, env_name: &str) -> Result<Self, ApiError> {
24        let ring = Self::new();
25        if raw.trim().is_empty() {
26            return Ok(ring);
27        }
28        let value: serde_json::Value = serde_json::from_str(raw)
29            .map_err(|e| ApiError::BadRequest(format!("{env_name}: {e}")))?;
30        match value {
31            serde_json::Value::Object(map) if map.contains_key("keys") => {
32                let keys = map.get("keys").and_then(|v| v.as_array()).ok_or_else(|| {
33                    ApiError::BadRequest(format!("{env_name}.keys must be array"))
34                })?;
35                for item in keys {
36                    let pk = item
37                        .get("public_key")
38                        .and_then(|v| v.as_str())
39                        .ok_or_else(|| ApiError::BadRequest("key missing public_key".into()))?;
40                    let material = item
41                        .get("secret_key_pem")
42                        .or_else(|| item.get("secret_key_path"))
43                        .and_then(|v| v.as_str())
44                        .ok_or_else(|| {
45                            ApiError::BadRequest(
46                                "key needs secret_key_pem or secret_key_path".into(),
47                            )
48                        })?;
49                    let pem = resolve_pem_material(material, env_name)?;
50                    ring.insert(pk.to_string(), pem);
51                }
52            }
53            serde_json::Value::Object(map) => {
54                for (pk, pem_v) in map {
55                    let material = pem_v.as_str().ok_or_else(|| {
56                        ApiError::BadRequest(format!(
57                            "{env_name} values must be PEM strings or paths"
58                        ))
59                    })?;
60                    let pem = resolve_pem_material(material, env_name)?;
61                    ring.insert(pk, pem);
62                }
63            }
64            _ => {
65                return Err(ApiError::BadRequest(format!(
66                    "{env_name} must be object map or {{keys:[...]}}"
67                )));
68            }
69        }
70        Ok(ring)
71    }
72
73    pub fn insert(&self, public_key: String, secret_key_pem: String) {
74        self.inner.write().insert(public_key, secret_key_pem);
75    }
76
77    #[must_use]
78    pub fn get(&self, public_key: &str) -> Option<String> {
79        self.inner.read().get(public_key).cloned()
80    }
81
82    pub fn require(&self, public_key: &str) -> Result<String, ApiError> {
83        self.get(public_key).ok_or_else(|| {
84            ApiError::NoSigner(format!(
85                "public_key {public_key} not found in local keyring"
86            ))
87        })
88    }
89
90    #[must_use]
91    pub fn len(&self) -> usize {
92        self.inner.read().len()
93    }
94
95    #[must_use]
96    pub fn is_empty(&self) -> bool {
97        self.len() == 0
98    }
99
100    #[must_use]
101    pub fn public_keys(&self) -> Vec<String> {
102        self.inner.read().keys().cloned().collect()
103    }
104}

106impl std::fmt::Debug for LocalKeyring {
107    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
108        f.debug_struct("LocalKeyring")
109            .field("len", &self.len())
110            .finish()
111    }

114/// Inline PEM text, or a path to a PEM file on disk.
115fn resolve_pem_material(value: &str, env_name: &str) -> Result<String, ApiError> {
116    let trimmed = value.trim();
117    if trimmed.is_empty() {
118        return Err(ApiError::BadRequest(format!(
119            "{env_name}: empty secret material"
120        )));
121    }
122    if looks_like_pem(trimmed) {
123        return Ok(trimmed.to_string());
124    }
125    let path = Path::new(trimmed);
126    if path.is_file() {
127        return std::fs::read_to_string(path).map_err(|e| {
128            ApiError::BadRequest(format!("{env_name}: read {}: {e}", path.display()))
129        });
130    }
131    Err(ApiError::BadRequest(format!(
132        "{env_name}: value is neither PEM text nor an existing file path"
133    )))
134}
135
136fn looks_like_pem(s: &str) -> bool {
137    s.contains("-----BEGIN") && s.contains("PRIVATE KEY")
138}

140#[cfg(test)]
141mod tests {
142    use super::*;
143    use std::io::Write;
144
145    #[test]
146    fn parse_map_and_keys_array() {
147        let map = r#"{"01aa":"-----BEGIN PRIVATE KEY-----\npem-a\n-----END PRIVATE KEY-----"}"#;
148        let ring = LocalKeyring::from_json(map, "LOCAL_KEYS_JSON").unwrap();
149        assert!(ring.get("01aa").unwrap().contains("pem-a"));
150
151        let arr = r#"{"keys":[{"public_key":"01bb","secret_key_pem":"-----BEGIN PRIVATE KEY-----\npem-b\n-----END PRIVATE KEY-----"}]}"#;
152        let ring = LocalKeyring::from_json(arr, "LOCAL_KEYS_JSON").unwrap();
153        assert!(ring.require("01bb").unwrap().contains("pem-b"));
154    }
155
156    #[test]
157    fn load_pem_from_path() {
158        let dir = tempfile::tempdir().unwrap();
159        let path = dir.path().join("secret.pem");
160        let mut f = std::fs::File::create(&path).unwrap();
161        writeln!(
162            f,
163            "-----BEGIN PRIVATE KEY-----\nfrom-file\n-----END PRIVATE KEY-----"
164        )
165        .unwrap();
166        drop(f);
167
168        let json = format!(
169            r#"{{"01cc":"{}"}}"#,
170            path.to_str().unwrap().replace('\\', "\\\\")
171        );
172        let ring = LocalKeyring::from_json(&json, "LOCAL_KEYS_JSON").unwrap();
173        assert!(ring.get("01cc").unwrap().contains("from-file"));
174
175        let json_arr = format!(
176            r#"{{"keys":[{{"public_key":"01dd","secret_key_path":"{}"}}]}}"#,
177            path.to_str().unwrap().replace('\\', "\\\\")
178        );
179        let ring = LocalKeyring::from_json(&json_arr, "LOCAL_KEYS_JSON_PRODUCTION").unwrap();
180        assert!(ring.require("01dd").unwrap().contains("from-file"));
181    }
