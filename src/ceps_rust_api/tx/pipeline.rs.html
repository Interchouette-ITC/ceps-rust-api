13#[derive(Debug, Clone, Serialize, ToSchema)]
14pub struct PipelineOutcome {
15    /// Deployed or returned transaction hash hex.
16    #[schema(example = "aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa")]
17    pub transaction_hash: String,
18    /// Signed Transaction JSON for put_transaction.
19    #[serde(skip_serializing_if = "Option::is_none")]
20    pub transaction: Option<Value>,
21    #[serde(skip_serializing_if = "Option::is_none")]
22    pub put_result: Option<Value>,
23    #[serde(skip_serializing_if = "Option::is_none")]
24    pub execution_result: Option<Value>,
25}
26
27impl PipelineOutcome {
28    #[must_use]
29    pub fn from_call(result: CallResult, submit: SubmitMode) -> Self {
30        let put_result = if result.put_result.is_null() {
31            None
32        } else {
33            Some(result.put_result)
34        };
35        let transaction = match submit {
36            SubmitMode::Return => result.transaction,
37            SubmitMode::Put => None,
38        };
39        Self {
40            transaction_hash: result.transaction_hash,
41            transaction,
42            put_result,
43            execution_result: result.execution_result,
44        }
45    }
46}

48/// Build client params for local PEM put/return or unsigned make-only.
49pub fn build_transaction_params(
50    state: &AppState,
51    envelope: &MutateEnvelope,
52) -> Result<TransactionParams, ApiError> {
53    envelope.validate_features()?;
54    let pk = envelope.signer.public_key.trim();
55
56    match envelope.submit {
57        SubmitMode::Return => {
58            if state.config.sign_backend.uses_local_keyring() {
59                if let Some(pem) = state.keyring.get(pk) {
60                    let mut tx = TransactionParams::new(pem, &envelope.payment_amount).make_only();
61                    tx = tx.with_chain_name(state.config.chain_name.clone());
62                    return Ok(tx);
63                }
64            }
65            let tx = TransactionParams::for_make(&envelope.payment_amount)
66                .with_initiator_addr(pk)
67                .with_chain_name(state.config.chain_name.clone());
68            Ok(tx)
69        }
70        SubmitMode::Put => match state.config.sign_backend {
71            SignBackend::None => Err(ApiError::NoSigner(
72                "submit=put requires SIGN_BACKEND local, local-production, or kms".into(),
73            )),
74            SignBackend::Local | SignBackend::LocalProduction => {
75                let pem = state.keyring.require(pk)?;
76                let mut tx = TransactionParams::new(pem, &envelope.payment_amount);
77                if !envelope.wait_on_put() {
78                    tx = tx.without_wait();
79                }
80                Ok(tx.with_chain_name(state.config.chain_name.clone()))
81            }
82            SignBackend::Kms => Ok(TransactionParams::for_make(&envelope.payment_amount)
83                .with_initiator_addr(pk)
84                .with_chain_name(state.config.chain_name.clone())),
85        },
86    }
87}

89/// After a CEP client call: for KMS put, sign+put+wait; otherwise map CallResult.
90pub async fn finalize_call(
91    state: &AppState,
92    core: &CEPClient,
93    envelope: &MutateEnvelope,
94    result: CallResult,
95) -> Result<PipelineOutcome, ApiError> {
96    envelope.validate_features()?;
97
98    if matches!(envelope.submit, SubmitMode::Put) && state.config.sign_backend == SignBackend::Kms {
99        #[cfg(feature = "sign-kms")]
100        {
101            let mut result = result;
102            let tx_json = result.transaction.take().ok_or_else(|| {
103                ApiError::Internal("make-only result missing transaction JSON".into())
104            })?;
105            let kms = state
106                .kms
107                .as_ref()
108                .ok_or_else(|| ApiError::Kms("KMS client not configured (set KMS_URL)".into()))?;
109            let signed = kms
110                .sign_transaction(&envelope.signer.public_key, &tx_json)
111                .await?;
112            let signed_tx = signed.get("transaction").cloned().unwrap_or(signed);
113            let wait = envelope.wait_on_put();
114            let result = crate::tx::put_signed_transaction(core, &signed_tx, wait).await?;
115            return Ok(PipelineOutcome::from_call(result, SubmitMode::Put));
116        }
117        #[cfg(not(feature = "sign-kms"))]
118        {
119            let _ = core;
120            return Err(ApiError::FeatureDisabled(
121                "SIGN_BACKEND=kms requires feature sign-kms".into(),
122            ));
123        }
124    }
125
126    #[cfg(not(feature = "sign-kms"))]
127    let _ = core;
128
129    Ok(PipelineOutcome::from_call(result, envelope.submit))
130}

151            signer: SignerRef {

154            payment_amount: "1".into(),
155        };
156        let err = build_transaction_params(&state, &env).unwrap_err();
